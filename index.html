window.sendUserAiMessage = async function() { const input = document.getElementById('aiChatInput'); const msg = input.value.trim(); if (!msg || !currentTripKey) return; const trip = liveTripData[currentTripKey]; trip.chatHistory = trip.chatHistory || []; trip.chatHistory.push({ id: "user_" + Date.now(), sender: "user", text: msg }); input.value = ''; renderAiChatHistory(); const container = document.getElementById('aiChatLogContainer'); const loadingId = 'ai_loading_' + Date.now(); container.innerHTML += `
提案・回答を考えています...
`; container.scrollTop = container.scrollHeight; const previousSuggestions = trip.chatHistory .filter(m => m.sender === 'ai' && m.suggestedSteps && m.suggestedSteps.length > 0) .map(m => m.suggestedSteps); const systemPrompt = `あなたは旅行しおりアプリ「シオリット」の優秀なAIコンシェルジュです。 旅行先: 『${trip.title}』 現在の日程・ステップ: ${JSON.stringify(trip.steps || [])} 過去に提案されたスケジュール履歴: ${JSON.stringify(previousSuggestions)} 【重要動作ルール】 1. 【元に戻す指示の場合（例: 「さっきの予定に戻して」「元に戻して」など）】: 過去のスケジュール履歴または直前の状態を特定し、元の予定リストを「suggestedSteps」にセットして「変更前のスケジュールを再提示いたしました！下のボタンから反映できます。」と案内してください。 2. 【お店・スポットの相談の場合】: 「おすすめ教えて」「お店提案して」「カフェある？」などの相談時は、必ず【2〜3件の候補（店名、ジャンル、特徴）】を箇条書きで比較提示してください。この時は「suggestedSteps」は含めずnullにしてください。 3. 【特定のスポット追加・変更の確定指示の場合】: 「〇番のお店にして」「〇〇を追加して」と確定した場合は、既存のスケジュールを綺麗に調整した新しい全ステップ配列を「suggestedSteps」に含めてください。 4. 【あいまいな企画展・イベント名の場合】: 「マリーアントワネット展」などのイベント名単体でも、旅行先エリアにある主要会場を推定して正式名称をmapQueryにセットしてスケジュール化してください。 【出力JSONフォーマット】 { "replyText": "ユーザーへの親身な回答メッセージ", "suggestedSteps": null または [全steps配列] }`; // ★ 余計な tools を一切含まない、元の安全な送信データ const payload = { contents: [{ parts: [{ text: `${systemPrompt}\n\nユーザーのメッセージ: ${msg}` }] }], generationConfig: { responseMimeType: "application/json", temperature: 0.7 } }; try { const res = await fetch(cloudflareWorkerEndpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (!res.ok) throw new Error("AIサーバー応答エラー (" + res.status + ")"); const result = await res.json(); const textResp = result.candidates?.[0]?.content?.parts?.[0]?.text; document.getElementById(loadingId)?.remove(); let replyDisplay = textResp; let newStepsCandidate = null; try { const cleanJsonStr = textResp.replace(/```json/gi, '').replace(/```/gi, '').trim(); const jsonMatch = cleanJsonStr.match(/\{[\s\S]*\}/); const parsed = JSON.parse(jsonMatch ? jsonMatch[0] : cleanJsonStr); if (parsed.replyText) replyDisplay = parsed.replyText; if (parsed.suggestedSteps && Array.isArray(parsed.suggestedSteps) && parsed.suggestedSteps.length > 0) { newStepsCandidate = parsed.suggestedSteps; } } catch(e) {} trip.chatHistory.push({ id: "ai_" + Date.now(), sender: "ai", text: replyDisplay, suggestedSteps: newStepsCandidate }); renderAiChatHistory(); await saveToCloud(); } catch(err) { document.getElementById(loadingId)?.remove(); trip.chatHistory.push({ id: "err_" + Date.now(), sender: "ai", text: `申し訳ありません。回答の生成中にエラーが発生しました（${err.message}）。もう一度お試しください。` }); renderAiChatHistory(); } };
